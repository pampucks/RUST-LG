iot_thing_unregistered()
Event for the unregistration of a registered thing. It is invoked whenever a registered thing is unregistered.
Defined in: iot.js. 
document.addEventListener(
     "iot_thing_unregistered",
     function (param) {
         // {String} param.thingUid - unregistered thing UID.
         console.log(
             "Event 'iot_thing_unregistered' is received.\n" +
             "thingUid = " + param.thingUid
         );
     },
     false
);
