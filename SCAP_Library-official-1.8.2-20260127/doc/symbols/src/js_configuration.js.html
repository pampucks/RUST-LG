 57                 if(commandURI.charAt(commandURI.length-1) != "/") {
 58                     commandURI += "/";
 59                 }
 60                 commandURI += reqParams.method;
 61 
 62                 var parameters = {};
 63                 if(reqParams.hasOwnProperty("parameters") === true)
 64                     parameters = reqParams.parameters;
 65 
 66                 var convertedRes = {};
 67 
 68                 var callback = function(res) {
 69                     console.log("res : " + JSON.stringify(res));
 70                     if (res.payload.returnValue === true) {
 71                         convertedRes = res.payload;
 72                         reqParams.onSuccess(convertedRes);
 73                     } else {
 74                         convertedRes.returnValue = false;
 75                         convertedRes.errorCode = res.payload.errorCode;
 76                         convertedRes.errorText = res.payload.errorText;
 77                         reqParams.onFailure(convertedRes);
 78                     }
 79                 }
 80                 if (service) {
 81                     service.call(commandURI, parameters, callback);
 82                 }
 83             };
 84         };
