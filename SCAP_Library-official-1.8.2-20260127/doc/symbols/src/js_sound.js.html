147      *      console.log("muted : " + cbObject.muted);

150      *      // Do something
151      *   }

153      *   function failureCb(cbObject) {
154      *      var errorCode = cbObject.errorCode;
155      *      var errorText = cbObject.errorText;
156      *      console.log ("Error Code [" + errorCode + "]: " + errorText);
157      *   }

159      *   var sound = new Sound();

167     Sound.prototype.getSoundStatus = function(successCallback, errorCallback) {
168         log("getSoundStatus: ");
169 
170         service.Request("luna://com.webos.service.commercial.scapadapter/sound", {
171             method: "getSoundStatus",
172             onSuccess: function(result) {
173                 if (result.returnValue === true) {
174                     delete result.returnValue;
175                     successCallback(result);
176                 }
177             },
178             onFailure: function(result) {
179                 delete result.returnValue;
180                 if (typeof errorCallback === 'function') {
181                     checkErrorCodeNText(result, "SGSS", "Sound.getSoundStatus returns failure.");
182                     errorCallback(result);
183                 }
184             }
185         });
186 
187         log("Sound.getSoundStatus Done");

503         service.Request("luna://com.webos.service.commercial.scapadapter/sound", {
504             method: "getSoundMode",
505             onSuccess: function(result) {
506                 if (result.returnValue === true) {
507                     delete result.returnValue;
508                     
509                     var convertToNum = ['balance'];
510                     for (var i = 0; i < convertToNum.length; i++) {
511                         var prop = convertToNum[i];
512                         if (prop in result) {
513                             var num = Number(result[prop]);
514                             result[prop] = !isNaN(num) ? num : result[prop];
515                         }
516                     }
517                     successCallback(result);
518                 }
519             },
520             onFailure: function(result) {
521                 delete result.returnValue;
522                 if (typeof errorCallback === 'function') {
523                     checkErrorCodeNText(result, "SGSM", "Sound.getSoundMode returns failure.");
524                     errorCallback(result);
525                 }
526             }
